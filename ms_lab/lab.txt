XQuery  – data.xml
sorting
for $book in doc("data.xml")/library/book
where xs:integer($book/published) > 2020
order by $book/title
return
    <book-summary id="{$book/@id}">
        {$book/title/text()}
    </book-summary>
Aggregation and Grouping
for $category in distinct-values(doc("data.xml")/library/book/@category)
let $books := doc("data.xml")/library/book[@category = $category]
return
    <category-summary>
        <category>{$category}</category>
        <book-count>{count($books)}</book-count>
        <average-price>
            {format-number(avg($books/price), "0.00")}
        </average-price>
    </category-summary>

distinct-values()
count()
avg()
format-number()
Conditional Transformation with FLWOR
for $book in doc("data.xml")/library/book
return
    <book-entry>
        <inventory-status>
            {
                if (xs:integer($book/stock) < 10)
                then "Reorder"
                else "In Stock"
            }
        </inventory-status>
        {$book/published}
        {$book/stock}
    </book-entry>
Custom XQuery Function
declare function local:price-in-inr(
    $price as xs:decimal,
    $currency as xs:string
) as xs:decimal {

    if ($currency = "GBP")
    then $price * 105.0
    else if ($currency = "EUR")
    then $price * 90.0
    else $price
};

for $book in doc("data.xml")/library/book
return
    <book-price>
        <title>{$book/title/text()}</title>
        <price-in-inr>
            {
                local:price-in-inr(
                    xs:decimal($book/price),
                    string($book/price/@currency)
                )
            }
        </price-in-inr>
    </book-price>
FLWOR Filtering and Ordering
for $book in doc("data.xml")/library/book
where
    ($book/@category = "technology"
     or $book/@category = "business")
    and xs:integer($book/published) >= 2020
order by xs:integer($book/stock) descending
return
    <book-summary>
        <title>{$book/title/text()}</title>
        <category>{$book/@category}</category>
        <stock>{$book/stock/text()}</stock>
    </book-summary>



Nested FLWOR and Category Summaries
for $category in distinct-values(doc("data.xml")/library/book/@category)
let $books := doc("data.xml")/library/book[@category = $category]
return
    <category-summary>
        <category>{$category}</category>
        <book-count>{count($books)}</book-count>
        <titles>
            {
                for $book in $books
                order by $book/title
                return <title>{$book/title/text()}</title>
            }
        </titles>
    </category-summary>

declare function local:price-in-inr(
    $price as xs:decimal,
    $currency as xs:string
) as xs:decimal {

    if ($currency = "GBP")
    then $price * 105.0
    else if ($currency = "EUR")
    then $price * 90.0
    else $price
};
