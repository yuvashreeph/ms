1. XML and XQuery question

Q1(a). XQuery using group by to group hospitals by type

xquery version "3.1";

for $h in //hospital
group by $type := $h/type
return
    <HospitalGroup type="{$type}">
    {
        for $hospital in $h
        return
            <hospital>
                <name>{$hospital/name}</name>
                <phone>{$hospital/phone}</phone>
                <address>{$hospital/address}</address>
            </hospital>
    }
    </HospitalGroup>


for $h in //hospital
group by $t := $h/type
return
<group type="{$t}">
{
    $h
}
</group>


Q1(b). FLWOR expression to extract all multispeciality hospitals
for $h in //hospital
where $h/type = "Multispeciality"
return $h


2. Docker and Containers
A. Scenario: Build arguments and environment variables

Step 1: Create app.js
const express = require("express");
const app = express();

const PORT = process.env.PORT || 3000;
const VERSION = process.env.VERSION || "1.0";

app.get("/", (req, res) => {
    res.send(`Node.js Application Version: ${VERSION}`);
});

app.listen(PORT, () => {
    console.log(`Server running on port ${PORT}`);
});

Step 2: Create package.json
{
  "name": "configurable-app",
  "version": "1.0.0",
  "main": "app.js",
  "scripts": {
    "start": "node app.js"
  },
  "dependencies": {
    "express": "^4.21.2"
  }
}

Step 3: Create Dockerfile
FROM node:20

WORKDIR /app

ARG APP_VERSION=1.0
ARG APP_PORT=3000

ENV VERSION=$APP_VERSION
ENV PORT=$APP_PORT

COPY package*.json ./

RUN npm install

COPY . .

EXPOSE 3000

CMD ["npm", "start"]

Step 4: Build and run
docker build --build-arg APP_VERSION=2.0 --build-arg APP_PORT=3000 -t myapp .
docker run -d -p 3000:3000 --name mycontainer myapp

docker build --build-arg VERSION=2.0 -t myapp .
docker run -d -p 3000:3000 myapp
docker ps
docker stop <container_id>

B. Scenario: Scale services horizontally using Docker Compose
Step 1: Create app.js
const express = require("express");
const app = express();

app.get("/", (req, res) => {
    res.send(`Request handled by container ${process.env.HOSTNAME}`);
});

app.listen(3000, () => {
    console.log("Server running on port 3000");
});

Step 2: Create package.json
{
  "name": "scalable-app",
  "version": "1.0.0",
  "main": "app.js",
  "scripts": {
    "start": "node app.js"
  },
  "dependencies": {
    "express": "^4.21.2"
  }
}

Step 3: Create Dockerfile
FROM node:20
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
EXPOSE 3000
CMD ["npm", "start"]

Step 4: Create docker-compose.yml
services:
  web:
    build: .
    expose:
      - "3000"

Step 5: Execute commands
docker compose up -d --build --scale web=3
docker compose ps
docker compose logs
docker compose down
